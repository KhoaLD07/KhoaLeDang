API HOOKING PRIMITIVE SOURCE

#include <windows.h>
#include <stdio.h>

int WINAPI HookedMessageBoxW(HWND hWnd, LPCWSTR lpText, LPCWSTR lpCaption, UINT uType) {
    wprintf(L"Blocked!\n");
    wprintf(L"Original text: %s\n", lpText);
    return IDOK;
}

void InstallHook() {
    HMODULE hUser32 = LoadLibraryA("user32.dll");
    FARPROC originalFuncAddress = GetProcAddress(hUser32, "MessageBoxW");

    if (originalFuncAddress == NULL) {
        printf("MessageBoxW not found!\n");
        return;
    }

    BYTE jmpInstruction[12] = { 0x48, 0xB8, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0xFF, 0xE0 };

    void* hookAddr = (void*)HookedMessageBoxW;
    memcpy(jmpInstruction + 2, &hookAddr, 8);

    DWORD oldProtect;
    VirtualProtect(originalFuncAddress, 12, PAGE_EXECUTE_READWRITE, &oldProtect);

    memcpy(originalFuncAddress, jmpInstruction, 12);

    VirtualProtect(originalFuncAddress, 12, oldProtect, &oldProtect);

    printf("Hook installed successfully!\n");
}

int main() {
    printf("--- Before Hook ---\n");
    MessageBoxW(NULL, L"Hello, this is a normal message box!", L"Test", MB_OK);

    InstallHook();

    printf("\n--- After Hook ---\n");
    MessageBoxW(NULL, L"You cannot see me!", L"Test", MB_OK);

    return 0;
}
